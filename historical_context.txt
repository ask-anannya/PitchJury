   150   const match = markdownText.match(/data:([^;]+);base64,([^)]+)/);
   151   if (!match) throw new Error("Could not parse Base64 image from response");
   152 
   153   const [, mimeType, base64Data] = match;
   154   const ext = mimeType.split("/")[1] ?? "jpg";
   155   const filePath = `uploads/${crypto.randomUUID()}.${ext}`;
   156 
   157   // Decode Base64 → Uint8Array
   158   const binaryStr = atob(base64Data);
   159   const bytes = new Uint8Array(binaryStr.length);
   160   for (let i = 0; i < binaryStr.length; i++) bytes[i] = binaryStr.charCodeAt(i);
   161 
   162   const { error } = await supabase.storage
   163     .from("generated-media")
   164     .upload(filePath, bytes, { contentType: mimeType, upsert: false });
   165 
   166   if (error) throw error;
   167 
   168   const { data: urlData } = supabase.storage.from("generated-media").getPublicUrl(filePath);
   169   return urlData.publicUrl;
   170 }
